#include <algorithm>


void preprocessImage(camera_fb_t* fb, TfLiteTensor* input) {
 
  const int model_width = 96;
  const int model_height = 96;
  

  uint8_t* original_image = fb->buf;
  int original_width = fb->width;
  int original_height = fb->height;
  
  // 1. Convert to grayscale if needed 
  
  // 2. Resize to model dimensions (96x96) using simple interpolation
  for (int y = 0; y < model_height; y++) {
    for (int x = 0; x < model_width; x++) {
      // Map new pixel coordinates to original image
      int orig_x = (x * original_width) / model_width;
      int orig_y = (y * original_height) / model_height;
      
      // Get grayscale value (assuming PIXFORMAT_GRAYSCALE)
      uint8_t pixel_value = original_image[orig_y * original_width + orig_x];
      
      // 3. Normalize to [0,1] range (change to [-1,1] if your model requires)
      float normalized_value = pixel_value / 255.0f;
      
      // Store in input tensor (adjust index based on your model's input format)
      input->data.f[y * model_width + x] = normalized_value;
    }
  }
}

void runInference(camera_fb_t* fb) {
  // Preprocess the image
  preprocessImage(fb, input);
  
  // Run inference
  TfLiteStatus invoke_status = interpreter->Invoke();
  if (invoke_status != kTfLiteOk) {
    Serial.println("Invoke failed");
    return;
  }

  // Process output
  processFOMOOutput(output);
}

void processFOMOOutput(TfLiteTensor* output) {
  const float confidence_threshold = 0.65; // Adjust based on your model
  int grid_width = output->dims->data[2];
  int grid_height = output->dims->data[1];
  
  for (int y = 0; y < grid_height; y++) {
    for (int x = 0; x < grid_width; x++) {
      int index = y * grid_width + x;
      float confidence = output->data.f[index];
      
      if (confidence > confidence_threshold) {
        // Convert grid coordinates back to original image coordinates
        int orig_x = (x * fb->width) / grid_width;
        int orig_y = (y * fb->height) / grid_height;
        
        Serial.printf("Detection at (%d,%d) - Confidence: %.2f\n", 
                     orig_x, orig_y, confidence);
      }
    }
  }
}
